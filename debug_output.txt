require_once '/home/dedjomo/Desktop/PortalNew/backend/config/database.php';

$database = new Database();
$db = $database->getConnection();

$email = 'bernabe.batchakui@portailmoodle.cm';

echo "--- Looking up Admin by email: $email ---\n";
$query = "SELECT * FROM administrators WHERE email = :email";
$stmt = $db->prepare($query);
$stmt->bindParam(':email', $email);
$stmt->execute();
$admin = $stmt->fetch(PDO::FETCH_ASSOC);

if ($admin) {
    echo "Admin Found:\n";
    echo "ID: " . $admin['id'] . "\n";
    echo "Name: " . $admin['first_name'] . " " . $admin['last_name'] . "\n";
    echo "Avatar URL: " . ($admin['avatar_url'] ? $admin['avatar_url'] : "NULL") . "\n";
    
    // Now search for instructor with similar name
    $fullName = $admin['first_name'] . " " . $admin['last_name']; // "Bernabe Batchakui" (Check order)
    // Or "Batchakui Bernabe" depending on stored fields.
    
    echo "\n--- Looking up Instructor by Name: '$fullName' or similar ---\n";
    $query2 = "SELECT * FROM instructors WHERE full_name LIKE :name";
    $stmt2 = $db->prepare($query2);
    $param = "%" . $admin['last_name'] . "%";
    $stmt2->bindParam(':name', $param);
    $stmt2->execute();
    $instructors = $stmt2->fetchAll(PDO::FETCH_ASSOC);
    
    if ($instructors) {
        foreach ($instructors as $ins) {
            echo "Instructor Found:\n";
            echo "ID: " . $ins['id'] . "\n";
            echo "Full Name: " . $ins['full_name'] . "\n";
            echo "Photo URL: " . ($ins['photo_url'] ? $ins['photo_url'] : "NULL") . "\n";
        }
    } else {
        echo "No instructor found matching last name '{$admin['last_name']}'.\n";
    }

} else {
    echo "Admin not found with email $email\n";
}
